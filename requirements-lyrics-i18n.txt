#      .\venv\Scripts\python.exe -m pip install -r requirements-lyrics-i18n.txt

#      .\venv\Scripts\python.exe -m pip install "ctc-forced-aligner @ git+https://github.com/MahmoudAshraf97/ctc-forced-aligner.git@64293cc6d711e57666c4a8b098e9fd93b381fd88"
